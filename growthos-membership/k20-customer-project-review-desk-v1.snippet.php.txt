            'reviewing'  => array('label' => 'در حال بررسی', 'tone' => 'amber'),
            'need_info'  => array('label' => 'نیاز به اطلاعات بیشتر', 'tone' => 'orange'),
            'ready_quote'=> array('label' => 'آماده پیش‌فاکتور', 'tone' => 'green'),

            'closed'     => array('label' => 'بسته شده', 'tone' => 'gray'),

if (!function_exists('k20_prd_now')) {
    function k20_prd_now() {
        return gmdate('c');
    }
}

if (!function_exists('k20_prd_capable')) {
    function k20_prd_capable() {
        return current_user_can('manage_woocommerce') || current_user_can('manage_options');
    }
}

if (!function_exists('k20_prd_member_state')) {
    function k20_prd_member_state($user_id) {
        $state = get_user_meta((int) $user_id, '_k20_member_value_v1', true);
        $state = is_array($state) ? $state : array();
        foreach (array('projects','calculations','proforma_refs','question_refs') as $key) {
            if (!isset($state[$key]) || !is_array($state[$key])) {
                $state[$key] = array();
            }
        }
        return $state;
    }
}

if (!function_exists('k20_prd_reviews')) {
    function k20_prd_reviews($user_id) {
        $reviews = get_user_meta((int) $user_id, '_k20_project_reviews_v1', true);
        return is_array($reviews) ? $reviews : array();
    }
}

if (!function_exists('k20_prd_save_reviews')) {
    function k20_prd_save_reviews($user_id, $reviews) {
        update_user_meta((int) $user_id, '_k20_project_reviews_v1', is_array($reviews) ? $reviews : array());
    }
}

if (!function_exists('k20_prd_project')) {
    function k20_prd_project($user_id, $project_id) {
        $state = k20_prd_member_state($user_id);
        foreach ($state['projects'] as $project) {
            if (is_array($project) && isset($project['id']) && hash_equals((string) $project['id'], (string) $project_id)) {
                return $project;
            }
        }
        return null;
    }
}

if (!function_exists('k20_prd_review')) {
    function k20_prd_review($user_id, $project_id) {
        $reviews = k20_prd_reviews($user_id);
        $row = isset($reviews[$project_id]) && is_array($reviews[$project_id]) ? $reviews[$project_id] : array();
        return wp_parse_args($row, array(
            'status' => 'new',
            'internal_note' => '',
            'customer_message' => '',
            'first_seen_at' => '',
            'updated_at' => '',
            'updated_by' => 0,
            'admin_notified_at' => '',
            'customer_notified_at' => '',
        ));
    }
}

if (!function_exists('k20_prd_status_label')) {
    function k20_prd_status_label($status) {
        $statuses = k20_prd_statuses();
        return isset($statuses[$status]) ? $statuses[$status]['label'] : $statuses['new']['label'];
    }
}

if (!function_exists('k20_prd_status_tone')) {
    function k20_prd_status_tone($status) {
        $statuses = k20_prd_statuses();
        return isset($statuses[$status]) ? $statuses[$status]['tone'] : 'blue';
    }
}

if (!function_exists('k20_prd_human_date')) {
    function k20_prd_human_date($value) {
        if (!$value) { return '—'; }
        $timestamp = strtotime((string) $value);
        return $timestamp ? wp_date('Y/m/d H:i', $timestamp) : '—';
    }
}

if (!function_exists('k20_prd_all_projects')) {
    function k20_prd_all_projects() {
        $user_ids = get_users(array(
            'meta_key' => '_k20_member_value_v1',
            'fields' => 'ID',
            'number' => 1000,
            'orderby' => 'ID',
            'order' => 'DESC',
        ));

        $rows = array();
        foreach ($user_ids as $user_id) {
            $state = k20_prd_member_state($user_id);
            $user = get_userdata($user_id);
            if (!$user) { continue; }

            foreach ($state['projects'] as $project) {
                if (!is_array($project) || empty($project['id'])) { continue; }
                $review = k20_prd_review($user_id, $project['id']);
                $rows[] = array(
                    'user_id' => (int) $user_id,
                    'customer_name' => $user->display_name ? $user->display_name : $user->user_login,
                    'project' => $project,
                    'review' => $review,
                );
            }
        }

        usort($rows, function($a, $b) {
            $at = !empty($a['project']['created_at']) ? strtotime($a['project']['created_at']) : 0;
            $bt = !empty($b['project']['created_at']) ? strtotime($b['project']['created_at']) : 0;
            return $bt <=> $at;
        });
        return $rows;
    }
}

if (!function_exists('k20_prd_handle_save')) {
    function k20_prd_handle_save() {
        if (!k20_prd_capable()) {
            wp_die('دسترسی کافی ندارید.');
        }

        $user_id = isset($_POST['user_id']) ? absint($_POST['user_id']) : 0;
        $project_id = isset($_POST['project_id']) ? sanitize_text_field(wp_unslash($_POST['project_id'])) : '';
        check_admin_referer('k20_prd_save_' . $user_id . '_' . $project_id);

        $project = k20_prd_project($user_id, $project_id);
        if (!$project) {
            wp_die('پروژه پیدا نشد.');
        }

        $statuses = k20_prd_statuses();
        $status = isset($_POST['status']) ? sanitize_key(wp_unslash($_POST['status'])) : 'new';
        if (!isset($statuses[$status])) { $status = 'new'; }

        $internal_note = isset($_POST['internal_note']) ? sanitize_textarea_field(wp_unslash($_POST['internal_note'])) : '';
        $customer_message = isset($_POST['customer_message']) ? sanitize_textarea_field(wp_unslash($_POST['customer_message'])) : '';
        if (function_exists('mb_substr')) {
            $internal_note = mb_substr($internal_note, 0, 5000);
            $customer_message = mb_substr($customer_message, 0, 3000);
        } else {
            $internal_note = substr($internal_note, 0, 5000);
            $customer_message = substr($customer_message, 0, 3000);
        }

        $reviews = k20_prd_reviews($user_id);
        $old = k20_prd_review($user_id, $project_id);
        $now = k20_prd_now();

        $reviews[$project_id] = array(
            'status' => $status,
            'internal_note' => $internal_note,
            'customer_message' => $customer_message,
            'first_seen_at' => $old['first_seen_at'] ? $old['first_seen_at'] : $now,
            'updated_at' => $now,
            'updated_by' => get_current_user_id(),
            'admin_notified_at' => $old['admin_notified_at'],
            'customer_notified_at' => $old['customer_notified_at'],
        );

        $send_email = !empty($_POST['notify_customer']) && $customer_message !== '';
        if ($send_email) {
            $customer = get_userdata($user_id);
            if ($customer && is_email($customer->user_email)) {
                $subject = 'وضعیت پروژه شما در کشاورز بیست به‌روزرسانی شد';
                $project_name = !empty($project['name']) ? $project['name'] : 'پروژه';
                $body = "پروژه: " . $project_name . "\n";
                $body .= "وضعیت: " . k20_prd_status_label($status) . "\n\n";
                $body .= $customer_message . "\n\n";
                $body .= "برای مشاهده جزئیات وارد حساب کاربری خود در کشاورز بیست شوید:\n";
                $body .= wc_get_page_permalink('myaccount');

                if (wp_mail($customer->user_email, $subject, $body)) {
                    $reviews[$project_id]['customer_notified_at'] = $now;
                }
            }
        }

        k20_prd_save_reviews($user_id, $reviews);

        wp_safe_redirect(k20_prd_admin_url(array(
            'view' => 'project',
            'user_id' => $user_id,
            'project_id' => rawurlencode($project_id),
            'saved' => '1',
        )));
        exit;
    }
    add_action('admin_post_k20_prd_save', 'k20_prd_handle_save');
}

if (!function_exists('k20_prd_new_project_event')) {
    function k20_prd_new_project_event($user_id, $event) {
        if ('project_saved' !== $event) { return; }

        $state = k20_prd_member_state($user_id);
        if (empty($state['projects'])) { return; }
        $project = end($state['projects']);
        if (!is_array($project) || empty($project['id'])) { return; }

        $reviews = k20_prd_reviews($user_id);
        $review = k20_prd_review($user_id, $project['id']);
        $now = k20_prd_now();

        if (!$review['first_seen_at']) {
            $review['first_seen_at'] = $now;
        }

        if (!$review['admin_notified_at']) {
            $admin_email = get_option('admin_email');
            if (is_email($admin_email)) {
                $subject = 'پروژه جدید در کشاورز بیست ثبت شد';
                $project_name = !empty($project['name']) ? $project['name'] : 'پروژه جدید';
                $body = "یک پروژه جدید با عنوان «" . $project_name . "» ثبت شده است.\n";
                $body .= "برای بررسی جزئیات وارد پنل پروژه‌های مشتریان شوید:\n";
                $body .= k20_prd_admin_url();
                if (wp_mail($admin_email, $subject, $body)) {
                    $review['admin_notified_at'] = $now;
                }
            }
        }

        $reviews[$project['id']] = $review;
        k20_prd_save_reviews($user_id, $reviews);
    }
    add_action('k20_member_value_event', 'k20_prd_new_project_event', 20, 2);
}

if (!function_exists('k20_prd_render_list')) {
    function k20_prd_render_list() {
        $rows = k20_prd_all_projects();
        $statuses = k20_prd_statuses();
        $filter_status = isset($_GET['status']) ? sanitize_key(wp_unslash($_GET['status'])) : '';
        $q = isset($_GET['q']) ? sanitize_text_field(wp_unslash($_GET['q'])) : '';

        $counts = array('all' => count($rows));
        foreach ($statuses as $key => $meta) { $counts[$key] = 0; }
        foreach ($rows as $row) {
            $s = isset($row['review']['status']) ? $row['review']['status'] : 'new';
            if (isset($counts[$s])) { $counts[$s]++; }
        }

        echo '<div class="k20-prd-stats">';
        foreach (array(
            'all' => 'کل پروژه‌ها',
            'new' => 'جدید',
            'reviewing' => 'در حال بررسی',
            'ready_quote' => 'آماده پیش‌فاکتور',
        ) as $key => $label) {
            echo '<div class="k20-prd-stat"><b>' . esc_html($counts[$key]) . '</b><span>' . esc_html($label) . '</span></div>';
        }
        echo '</div>';

        echo '<form class="k20-prd-toolbar" method="get">';
        echo '<input type="hidden" name="page" value="k20-project-desk">';
        echo '<input type="search" name="q" value="' . esc_attr($q) . '" placeholder="جست‌وجوی پروژه، مشتری، کشت یا استان">';
        echo '<select name="status"><option value="">همه وضعیت‌ها</option>';
        foreach ($statuses as $key => $meta) {
            echo '<option value="' . esc_attr($key) . '" ' . selected($filter_status, $key, false) . '>' . esc_html($meta['label']) . '</option>';
        }
        echo '</select>';
        echo '<button class="button button-primary" type="submit">اعمال فیلتر</button>';
        echo '<a class="button" href="' . esc_url(k20_prd_admin_url()) . '">پاک‌کردن فیلتر</a>';
        echo '</form>';

        $filtered = array_values(array_filter($rows, function($row) use ($filter_status, $q) {
            if ($filter_status && $row['review']['status'] !== $filter_status) { return false; }
            if (!$q) { return true; }
            $p = $row['project'];
            $haystack = implode(' ', array(
                $row['customer_name'],
                isset($p['name']) ? $p['name'] : '',
                isset($p['crop']) ? $p['crop'] : '',
                isset($p['province']) ? $p['province'] : '',
            ));
            if (function_exists('mb_stripos')) { return false !== mb_stripos($haystack, $q); }
            return false !== stripos($haystack, $q);
        }));

        $per_page = 20;
        $paged = isset($_GET['paged']) ? max(1, absint($_GET['paged'])) : 1;
        $total = count($filtered);
        $page_rows = array_slice($filtered, ($paged - 1) * $per_page, $per_page);

        echo '<table class="k20-prd-table"><thead><tr>';
        echo '<th>پروژه</th><th>مشتری</th><th>کشت</th><th>مساحت</th><th>تاریخ ثبت</th><th>وضعیت</th><th></th>';
        echo '</tr></thead><tbody>';
        if (!$page_rows) {
            echo '<tr><td colspan="7">پروژه‌ای با این فیلتر پیدا نشد.</td></tr>';
        }
        foreach ($page_rows as $row) {
            $p = $row['project'];
            $status = $row['review']['status'];
            $tone = k20_prd_status_tone($status);
            $url = k20_prd_admin_url(array(
                'view' => 'project',
                'user_id' => $row['user_id'],
                'project_id' => rawurlencode($p['id']),
            ));
            echo '<tr>';
            echo '<td><strong>' . esc_html(isset($p['name']) ? $p['name'] : 'بدون عنوان') . '</strong><br><small>' . esc_html($p['id']) . '</small></td>';
            echo '<td>' . esc_html($row['customer_name']) . '</td>';
            echo '<td>' . esc_html(isset($p['crop']) && $p['crop'] ? $p['crop'] : '—') . '</td>';
            echo '<td>' . esc_html(isset($p['area_ha']) && $p['area_ha'] !== null ? $p['area_ha'] . ' هکتار' : '—') . '</td>';
            echo '<td>' . esc_html(k20_prd_human_date(isset($p['created_at']) ? $p['created_at'] : '')) . '</td>';
            echo '<td><span class="k20-badge k20-tone-' . esc_attr($tone) . '">' . esc_html(k20_prd_status_label($status)) . '</span></td>';
            echo '<td><a class="button" href="' . esc_url($url) . '">بررسی پروژه</a></td>';
            echo '</tr>';
        }
        echo '</tbody></table>';

        if ($total > $per_page) {
            $base = k20_prd_admin_url(array(
                'status' => $filter_status,
                'q' => $q,
                'paged' => '%#%',
            ));
            echo '<div class="tablenav"><div class="tablenav-pages">';
            echo wp_kses_post(paginate_links(array(
                'base' => $base,
                'format' => '',
                'current' => $paged,
                'total' => ceil($total / $per_page),
                'prev_text' => 'قبلی',
                'next_text' => 'بعدی',
            )));
            echo '</div></div>';
        }
    }
}

if (!function_exists('k20_prd_render_project')) {
    function k20_prd_render_project() {
        $user_id = isset($_GET['user_id']) ? absint($_GET['user_id']) : 0;
        $project_id = isset($_GET['project_id']) ? sanitize_text_field(wp_unslash($_GET['project_id'])) : '';
        $project = k20_prd_project($user_id, $project_id);
        $customer = get_userdata($user_id);

        if (!$project || !$customer) {
            echo '<div class="notice notice-error"><p>پروژه پیدا نشد.</p></div>';
            echo '<p><a class="button" href="' . esc_url(k20_prd_admin_url()) . '">بازگشت</a></p>';
            return;
        }

        $review = k20_prd_review($user_id, $project_id);
        $state = k20_prd_member_state($user_id);
        $statuses = k20_prd_statuses();

        if (!empty($_GET['saved'])) {
            echo '<div class="notice-inline">تغییرات پروژه ذخیره شد.</div>';
        }

        echo '<p><a href="' . esc_url(k20_prd_admin_url()) . '">← بازگشت به همه پروژه‌ها</a></p>';
        echo '<div class="k20-prd-grid">';

        echo '<div>';
        echo '<div class="k20-prd-card"><h2>' . esc_html($project['name']) . '</h2>';
        echo '<div class="k20-prd-fields">';
        $fields = array(
            'مشتری' => $customer->display_name ? $customer->display_name : $customer->user_login,
            'کشت' => isset($project['crop']) && $project['crop'] ? $project['crop'] : '—',

            'منبع آب' => isset($project['water_source']) && $project['water_source'] ? $project['water_source'] : '—',
            'فشار تقریبی' => isset($project['pressure_bar']) && $project['pressure_bar'] !== null ? $project['pressure_bar'] . ' bar' : '—',
            'تاریخ ثبت' => k20_prd_human_date(isset($project['created_at']) ? $project['created_at'] : ''),
            'شناسه پروژه' => $project['id'],
        );
        foreach ($fields as $label => $value) {
            echo '<div class="k20-prd-field"><small>' . esc_html($label) . '</small><strong>' . esc_html($value) . '</strong></div>';
        }
        echo '</div></div>';

        echo '<div class="k20-prd-card"><h3>سوابق مرتبط مشتری</h3>';
        echo '<p><strong>محاسبات ذخیره‌شده:</strong> ' . esc_html(count($state['calculations'])) . '</p>';
        echo '<p><strong>پرسش‌های فنی:</strong> ' . esc_html(count($state['question_refs'])) . '</p>';
        echo '<p><strong>پیش‌فاکتورهای ثبت‌شده:</strong> ' . esc_html(count($state['proforma_refs'])) . '</p>';
        echo '</div>';
        echo '</div>';

        echo '<div class="k20-prd-card"><h2>بررسی کارشناس</h2>';
        echo '<form method="post" action="' . esc_url(admin_url('admin-post.php')) . '">';
        echo '<input type="hidden" name="action" value="k20_prd_save">';
        echo '<input type="hidden" name="user_id" value="' . esc_attr($user_id) . '">';
        echo '<input type="hidden" name="project_id" value="' . esc_attr($project_id) . '">';
        wp_nonce_field('k20_prd_save_' . $user_id . '_' . $project_id);

        echo '<label for="k20-prd-status">وضعیت پروژه</label>';
        echo '<select id="k20-prd-status" name="status">';
        foreach ($statuses as $key => $meta) {
            echo '<option value="' . esc_attr($key) . '" ' . selected($review['status'], $key, false) . '>' . esc_html($meta['label']) . '</option>';
        }
        echo '</select>';

        echo '<label for="k20-prd-internal">یادداشت داخلی مدیر</label>';
        echo '<textarea id="k20-prd-internal" name="internal_note" maxlength="5000" placeholder="این متن فقط در پنل مدیریت دیده می‌شود.">' . esc_textarea($review['internal_note']) . '</textarea>';

        echo '<label for="k20-prd-customer">پاسخ قابل‌نمایش به مشتری</label>';
        echo '<textarea id="k20-prd-customer" name="customer_message" maxlength="3000" placeholder="نتیجه بررسی، اطلاعات موردنیاز یا توضیحی که مشتری باید در حساب خود ببیند.">' . esc_textarea($review['customer_message']) . '</textarea>';

        echo '<p><label style="font-weight:700"><input type="checkbox" name="notify_customer" value="1"> بعد از ذخیره، همین پاسخ به ایمیل مشتری هم اطلاع داده شود.</label></p>';
        echo '<p><button class="button button-primary button-large" type="submit">ذخیره بررسی و پاسخ</button></p>';

        if ($review['updated_at']) {
            echo '<hr><p><small>آخرین به‌روزرسانی: ' . esc_html(k20_prd_human_date($review['updated_at'])) . '</small></p>';
        }
        if ($review['customer_notified_at']) {
            echo '<p><small>آخرین اعلان ایمیلی موفق: ' . esc_html(k20_prd_human_date($review['customer_notified_at'])) . '</small></p>';
        }

        echo '</form></div>';
        echo '</div>';
    }

if (!function_exists('k20_prd_customer_board')) {
    function k20_prd_customer_board() {
        if (!is_user_logged_in()) { return; }
        $user_id = get_current_user_id();
        $state = k20_prd_member_state($user_id);
        if (empty($state['projects'])) { return; }

        echo '<section class="k20-project-status-board" aria-labelledby="k20-project-status-title">';
        echo '<h2 id="k20-project-status-title">وضعیت بررسی پروژه‌ها</h2>';
        echo '<p>آخرین وضعیت و پاسخ کارشناس کشاورز بیست برای پروژه‌های ذخیره‌شده شما.</p>';
        echo '<div class="k20-project-status-list">';

        foreach (array_reverse($state['projects']) as $project) {
            if (!is_array($project) || empty($project['id'])) { continue; }
            $review = k20_prd_review($user_id, $project['id']);
            $status = $review['status'];
            $tone = k20_prd_status_tone($status);

            echo '<article class="k20-project-status-card">';
            echo '<div class="k20-project-status-head"><strong>' . esc_html(isset($project['name']) ? $project['name'] : 'پروژه') . '</strong>';
            echo '<span class="k20-badge k20-tone-' . esc_attr($tone) . '">' . esc_html(k20_prd_status_label($status)) . '</span></div>';

            echo '<div class="k20-project-status-meta">';
            if (!empty($project['crop'])) { echo '<span>کشت: ' . esc_html($project['crop']) . '</span>'; }
            if (isset($project['area_ha']) && $project['area_ha'] !== null) { echo '<span>مساحت: ' . esc_html($project['area_ha']) . ' هکتار</span>'; }
            if ($review['updated_at']) { echo '<span>آخرین بررسی: ' . esc_html(k20_prd_human_date($review['updated_at'])) . '</span>'; }
            echo '</div>';

            if ($review['customer_message'] !== '') {
                echo '<div class="k20-project-status-message"><strong>پاسخ کارشناس:</strong><br>' . nl2br(esc_html($review['customer_message'])) . '</div>';
            } elseif ('new' === $status) {
                echo '<div class="k20-project-status-message">پروژه دریافت شده و در صف بررسی کارشناسی قرار دارد.</div>';
            } elseif ('reviewing' === $status) {
                echo '<div class="k20-project-status-message">پروژه شما در حال بررسی کارشناسی است.</div>';
            }

            echo '</article>';
        }

        echo '</div></section>';
    }
    add_action('woocommerce_account_dashboard', 'k20_prd_customer_board', 36);
}
