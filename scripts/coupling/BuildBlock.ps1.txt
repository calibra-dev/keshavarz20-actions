function Build-RecommendationBlock($Target) {
  $sizes=@($Target.sizes_mm)
  $a=Convert-ToPersianDigits([string]$sizes[0])
  if([string]$Target.kind -eq 'reducer'){
    $b=Convert-ToPersianDigits([string]$sizes[1])
    $heading="قطعات پیشنهادی برای تکمیل خط پلی‌اتیلن $a به $b میلی‌متر"
    $intro="این رابط تبدیل بین دو سایز $a و $b میلی‌متر قرار می‌گیرد. برای اینکه سفارش ناقص نماند، پیشنهادهای زیر بر اساس هر دو سمت اتصال و فقط از محصولات واقعی فروشگاه انتخاب شده‌اند. هر سمت را جداگانه با سایز درج‌شده روی لوله یا اتصال موجود تطبیق دهید."
  } else {
    $heading="قطعات پیشنهادی برای تکمیل خط پلی‌اتیلن $a میلی‌متر"
    $intro="این رابط مساوی برای اتصال دو بخش هم‌سایز خط پلی‌اتیلن $a میلی‌متری انتخاب می‌شود. برای تکمیل سفارش، لوله و چند قطعه کاربردی همان سایز از محصولات واقعی فروشگاه در ادامه آمده‌اند."
  }
  $items=@()
  foreach($r in @($Target.proposed_products)){
    $s=Convert-ToPersianDigits([string]$r.side_mm)
    $label=switch([string]$r.family){'pipe'{"لوله پلی‌اتیلن $s میلی‌متر"}'coupling'{"رابط مساوی $s میلی‌متر"}'endcap'{"درپوش انتهایی $s میلی‌متر"}'elbow'{"زانو $s میلی‌متر"}'tee'{"سه‌راه $s میلی‌متر"}'valve'{"شیر $s میلی‌متر"}default{"اتصال $s میلی‌متر"}}
    $note=switch([string]$r.family){'pipe'{"برای سمت $s میلی‌متری خط."}'coupling'{'برای ادامه مستقیم همین سایز در بخش دیگری از خط.'}'endcap'{'برای بستن انتهای همان شاخه.'}'elbow'{'برای تغییر مسیر همان شاخه.'}'tee'{'برای ایجاد انشعاب همان شاخه.'}'valve'{'برای قطع و وصل جریان همان شاخه.'}default{'برای تکمیل همان سمت خط.'}}
    $name=[Net.WebUtility]::HtmlEncode([string]$r.name);$url=[Net.WebUtility]::HtmlEncode([string]$r.permalink)
    $items+=('<li style="margin:7px 0"><strong>{0}:</strong> <a style="color:#176b3a;font-weight:700" href="{1}">{2}</a> — {3}</li>' -f $label,$url,$name,$note)
  }
  $missing=@($Target.missing_pipe_sizes);$missingText=''
  if($missing.Count){$m=@($missing|ForEach-Object{Convert-ToPersianDigits([string]$_)})-join' و ';$missingText="<p><strong>نکته:</strong> برای سایز $m میلی‌متر، لوله پلی‌اتیلن منتشرشده‌ای در دسته لوله‌های فروشگاه پیدا نشد؛ بنابراین لینک ساختگی یا سایز نامرتبط پیشنهاد نشده است.</p>"}
  $list=$items-join"`n"
  return @"
<h2 style="font-size:24px;line-height:1.8;color:#176b3a;margin:34px 0 14px;border-right:5px solid #5d9a68;padding-right:12px">$heading</h2>
<div style="background:#f7fbf8;border:1px solid #dbe8df;border-radius:16px;padding:17px;margin:15px 0">
<p>$intro</p>
<ul style="padding-right:22px;margin:10px 0">$list</ul>
$missingText
<p style="margin:12px 0 0"><strong>قبل از خرید:</strong> مبنا را سایز میلی‌متری لوله و دو سر اتصال قرار دهید. اگر روی خط فعلی چند عدد یا سایز متفاوت دیده می‌شود، هر دو سمت را جداگانه بررسی کنید.</p>
</div>
"@
}
